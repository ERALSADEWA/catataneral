# ==========================================
# CHR INTERFACE RECOVERY SCRIPT
# WAN  = ether1
# LAN  = ether2
# LAN IP = 192.168.19.1/24
# ==========================================

# 1. NORMALISASI NAMA INTERFACE
/interface ethernet
set [find where default-name="ether1"] name=ether1
set [find where default-name="ether2"] name=ether2
set [find where default-name="ether3"] name=ether3
set [find where default-name="ether4"] name=ether4

# 2. PERBAIKI IP ADDRESS LAN
/ip address
:if ([:len [find where address="192.168.19.1/24"]] > 0) do={
    set [find where address="192.168.19.1/24"] interface=ether2
} else={
    add address=192.168.19.1/24 interface=ether2 comment="LAN Gateway"
}

# 3. PERBAIKI DHCP CLIENT WAN
/ip dhcp-client
set [find where name="client1"] interface=ether1 disabled=no

# 4. PERBAIKI DHCP SERVER LAN
/ip dhcp-server
set [find where name="dhcp1"] interface=ether2 disabled=no

# 5. PERBAIKI FIREWALL NAT
/ip firewall nat
:if ([:len [find where chain=srcnat and action=masquerade]] > 0) do={
    set [find where chain=srcnat and action=masquerade] out-interface=ether1
} else={
    add chain=srcnat action=masquerade out-interface=ether1 comment="WAN Masquerade"
}

# 6. SELESAI
:put "================================"
:put "INTERFACE RECOVERY COMPLETED"
:put "WAN  : ether1"
:put "LAN  : ether2"
:put "================================"
